# REFIM: the CPU miner (refined EFIM). Builds standalone (cmake -S REFIM) or via the root CMakeLists.txt.
cmake_minimum_required(VERSION 3.24)
project(REFIM LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_CONFIGURATION_TYPES AND NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

include(FetchContent)
# Pin dependency commits for reproducible builds (CLI11 2.6.2, spdlog 1.16.0).
if(NOT TARGET CLI11::CLI11)
    FetchContent_Declare(CLI11
        GIT_REPOSITORY https://github.com/CLIUtils/CLI11.git
        GIT_TAG 37bb6edc5317e99af72ef48405e65d9ca5218861)
    FetchContent_MakeAvailable(CLI11)
endif()
if(NOT TARGET spdlog::spdlog)
    FetchContent_Declare(spdlog
        GIT_REPOSITORY https://github.com/gabime/spdlog.git
        GIT_TAG 486b55554f11c9cccc913e11a87085b2a91f706f)
    FetchContent_MakeAvailable(spdlog)
endif()
find_package(Threads REQUIRED)

# Sources: io/ reads the dataset and writes itemsets, mine/ is the search, sys/ holds NUMA,
# allocation and threading helpers.
add_executable(refim
    src/main.cpp
    src/app/options.cpp
    src/app/logging.cpp
    src/app/run.cpp
    src/io/reader.cpp
    src/io/reader/text.cpp
    src/io/reader/name_table.cpp
    src/io/reader/read_all.cpp
    src/io/reader/split_into_chunks.cpp
    src/io/reader/parse_chunks.cpp
    src/io/reader/merge_names.cpp
    src/io/reader/rank_items.cpp
    src/io/reader/rename_transactions.cpp
    src/io/reader/check_utilities.cpp
    src/mine/refim.cpp
    src/mine/simd_mode.cpp
    src/sys/alloc.cpp
    src/sys/numa.cpp)
target_include_directories(refim PRIVATE include ${CMAKE_CURRENT_BINARY_DIR}/generated)
target_link_libraries(refim PRIVATE Threads::Threads CLI11::CLI11 spdlog::spdlog)
set(REFIM_COMPILE_OPTIONS -Wall -Wextra $<$<CONFIG:Release>:-O3 -march=native>)
target_compile_options(refim PRIVATE ${REFIM_COMPILE_OPTIONS})

# The search (and the root database it starts from) is compiled once per (item bits, utility bits) pair, each
# copy in its own namespace (include/mine/widths.hpp); Refim::mine() picks the narrowest pair that holds the
# dataset, or --key-bits/--util-bits. Fewer pairs build faster: -DREFIM_WIDTHS="32:64" builds one.
set(REFIM_WIDTHS "8:16;8:32;8:64;16:16;16:32;16:64;32:16;32:32;32:64" CACHE STRING
    "item:utility bit widths to build the search for (items 8/16/32, utilities 16/32/64)")
set(REFIM_WIDTH_SOURCES
    src/io/root.cpp
    src/io/reader/concatenate.cpp
    src/io/reader/sort_transactions.cpp
    src/io/reader/build_root.cpp
    src/io/reader/find_primary.cpp
    src/io/output.cpp
    src/mine/search.cpp
    src/mine/worker/worker.cpp
    src/mine/worker/expand.cpp
    src/mine/worker/leaf_utility.cpp
    src/mine/worker/project_child.cpp
    src/mine/worker/record_pattern.cpp
    src/mine/worker/select_extensions.cpp
    src/mine/worker/compact_child.cpp
    src/mine/worker/descend.cpp
    src/mine/simd.cpp)
set(REFIM_BUILT_WIDTHS "")
set(REFIM_VARIANTS "")
foreach(widths IN LISTS REFIM_WIDTHS)
    if(NOT widths MATCHES "^(8|16|32):(16|32|64)$")
        message(FATAL_ERROR "REFIM_WIDTHS: '${widths}' is not item:utility with items 8/16/32 and utilities 16/32/64")
    endif()
    set(variant refim_k${CMAKE_MATCH_1}_u${CMAKE_MATCH_2})
    add_library(${variant} OBJECT ${REFIM_WIDTH_SOURCES})
    target_include_directories(${variant} PRIVATE include ${CMAKE_CURRENT_BINARY_DIR}/generated)
    target_link_libraries(${variant} PRIVATE Threads::Threads spdlog::spdlog)
    target_compile_options(${variant} PRIVATE ${REFIM_COMPILE_OPTIONS})
    target_compile_definitions(${variant} PRIVATE REFIM_KEY_BITS=${CMAKE_MATCH_1} REFIM_UTIL_BITS=${CMAKE_MATCH_2})
    target_link_libraries(refim PRIVATE ${variant})
    list(APPEND REFIM_VARIANTS ${variant})
    string(APPEND REFIM_BUILT_WIDTHS " X(${CMAKE_MATCH_1}, ${CMAKE_MATCH_2})")
endforeach()
if(REFIM_BUILT_WIDTHS STREQUAL "")
    message(FATAL_ERROR "REFIM_WIDTHS is empty")
endif()
file(CONFIGURE OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/generated/mine/built_widths.hpp CONTENT
"#pragma once
// Generated by CMake from REFIM_WIDTHS: X(item bits, utility bits) for every search built.
#define REFIM_FOR_EACH_BUILT_WIDTH(X)${REFIM_BUILT_WIDTHS}
")

# refim build option: link-time optimisation.
option(REFIM_LTO "Link-time optimisation for refim" ON)
if(CMAKE_BUILD_TYPE STREQUAL "Release" AND REFIM_LTO)
    foreach(target IN ITEMS refim ${REFIM_VARIANTS})
        target_compile_options(${target} PRIVATE -flto=auto -fno-fat-lto-objects)
    endforeach()
    target_link_options(refim PRIVATE -flto=auto)
endif()
