36static char getSeparator(const std::string &sepStr)
37{
38 if (sepStr == "\\s")
39 return ' ';
40 if (sepStr == "\\t")
41 return '\t';
42 if (sepStr == "\\n")
43 return '\n';
44 if (sepStr == ",")
45 return ',';
46 if (sepStr == ";")
47 return ';';
48 if (sepStr == "|")
49 return '|';
50
51 // Default separator (fallback to space)
52 std::cerr << "Unknown separator: " << sepStr << ". Exiting.\n";
53 std::exit(1);
54}

157 void parse(int argc, char *argv[])
158 {
159 try
160 {
161 if (argc > 0)
162 program_name = argv[0];
163
164 for (int i = 1; i < argc; ++i)
165 {
166 std::string arg = argv[i];
167
168 // Check if the argument matches a known flag or argument
169 if (args.find(arg) != args.end())
170 {
171 // It's a flag
172 if (args[arg].is_flag)
173 {
174 args[arg].value = "true";
175 }
176 // It's a regular argument
177 else if (i + 1 < argc)
178 {
179 args[arg].value = argv[++i];
180 }
181 // Value missing
182 else
183 {
184 throw std::runtime_error("Missing value for argument: " + arg);
185 }
186 }
187 else if (arg == "--help" || arg == "-h")
188 {
189 print_help();
190 std::exit(0);
191 }
192 else
193 {
194 // Anything that doesn't match a known argument/flag is treated as positional
195 positional_args.push_back(arg);
196 }
197 }
198
199 // Validate required arguments
200 for (const auto &[key, arg] : args)
201 {
202 if (arg.required && !arg.value.has_value())
203 {
204 throw std::runtime_error("Missing required argument: " + key);
205 }
206 }
207
208 // Apply default values where needed
209 for (auto &[key, arg] : args)
210 {
211 if (!arg.value.has_value() && arg.default_value.has_value())
212 {
213 arg.value = arg.default_value;
214 }
215 }
216 }
217 catch (const std::exception &e)
218 {
219 std::cerr << "Error: " << e.what() << "\n\n";
220 print_help();
221 std::exit(1);
